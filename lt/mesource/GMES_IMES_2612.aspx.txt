        function SetProdQtyDiff() {
            try {

                ucProductRealgrid_gridView.commit(true);

                var dataArray = ucProductRealgrid_dataProvider.getJsonRows(0, -1);
                var GroupbyData = [];
                for (var i = dataArray.length; i >= 0; i--) {
                    var index = i - 1;
                    if (index >= 0) {
                        var data = dataArray[index];

                        let QTY_ERP_IN = 0;
                        let QTY_IN_DIFF = 0;

                        let QTY_ERP_OT = 0;
                        let QTY_OT_DIFF = 0;

                        let DIGIT = 0.01;

                        let YIELD_GOAL = (data.YIELD_GOAL == null || data.YIELD_GOAL == undefined) ? 1 : data.YIELD_GOAL * DIGIT;
                        let RATIO_OT = (data.RATIO_OT == null || data.RATIO_OT == undefined) ? 1 : data.RATIO_OT * DIGIT;

                        if (data.OT_EDIT == "Y") {
                            //출고 확정량
                            //출고량 + 출고차이량

                            if (data.QTY_ERP_OT == null || data.QTY_ERP_OT == undefined)
                                QTY_ERP_OT = 0;
                            else
                                QTY_ERP_OT = data.QTY_ERP_OT;

                            if (data.QTY_OT_DIFF == null || data.QTY_OT_DIFF == undefined)
                                QTY_OT_DIFF = 0;
                            else
                                QTY_OT_DIFF = data.QTY_OT_DIFF;

                            //출고 차이량 = (입고확정량/목표수율*배부비율)-출고량(SUM)
                            if (YIELD_GOAL <= 0) YIELD_GOAL = 1;

                            let OT_QTY = 0;
                            if (RATIO_OT == 1) {
                                OT_QTY = data.QTY_ERP_OT_SUM;
                            } else {
                                OT_QTY = data.QTY_ERP_OT;
                            }

                            /*OT_QTY = data.QTY_ERP_OT;*/

                            if (data.QTY_CFM_IN <= 0) {
                                data.QTY_OT_DIFF = 0;
                            } else {
                                data.QTY_OT_DIFF = (data.QTY_CFM_IN / YIELD_GOAL.toFixed(10) * RATIO_OT) - OT_QTY;
                            }

                            //출고 확정량
                            data.QTY_CFM_OT = QTY_ERP_OT + data.QTY_OT_DIFF;

                            dataArray[index].QTY_OT_DIFF = data.QTY_OT_DIFF;
                            dataArray[index].QTY_CFM_OT = data.QTY_CFM_OT;

                            ucProductRealgrid_gridView.setValue(index, "QTY_OT_DIFF", dataArray[index].QTY_OT_DIFF); //출고 차이량
                            ucProductRealgrid_gridView.setValue(index, "QTY_CFM_OT", dataArray[index].QTY_CFM_OT); //출고 확정량
                            ucProductRealgrid_gridView.commit(true);
                            //추적 가능한 입고 정보 조회
                            var dataArrayData = $.grep(dataArray, function (el, l) {
                                return el.EQSGID == data.EQSGID_OT && el.PROCID == data.PROCID_OT && el.PRODID == data.MTRLID;
                            });

                            if (dataArrayData.length > 0) {

                                GridProdSetValue(dataArrayData[0], data.QTY_OT_DIFF);
                            }

                        }
                    }
                }

                ucProductRealgrid_gridView.commit(true);

                //확정 수율 계산
                dataArray = ucProductRealgrid_dataProvider.getJsonRows(0, -1);

                let EQSGID = '';
                let PRODID = '';
                let PROCID = '';
                let YIELD_CFM = 0;

                let ProdGrouby = [];
                for (var i = 0; i < dataArray.length; i++) {
                    if (EQSGID == dataArray[i].EQSGID
                        && PROCID == dataArray[i].PROCID
                        && PRODID == dataArray[i].PRODID) {
                        var dataArrayData = $.grep(ProdGrouby, function (el, l) {
                            return el.EQSGID == dataArray[i].EQSGID
                                && el.PROCID == dataArray[i].PROCID
                                && el.PRODID == dataArray[i].PRODID;
                        });

                        if (dataArrayData.length > 0) {
                            dataArrayData[0].QTY_CFM_OT_SUM += dataArray[i].QTY_CFM_OT;
                        }
                    } else {
                        EQSGID = dataArray[i].EQSGID;
                        PROCID = dataArray[i].PROCID;
                        PRODID = dataArray[i].PRODID;

                        ProdGrouby.push({
                            EQSGID: dataArray[i].EQSGID
                            , PROCID: dataArray[i].PROCID
                            , PRODID: dataArray[i].PRODID
                            , QTY_CFM_OT_SUM: dataArray[i].QTY_CFM_OT
                        });
                    }
                }

                for (var i = 0; i < dataArray.length; i++) {

                    var dataArrayData = $.grep(ProdGrouby, function (el, l) {
                        return el.EQSGID == dataArray[i].EQSGID
                            && el.PROCID == dataArray[i].PROCID
                            && el.PRODID == dataArray[i].PRODID;
                    });

                    if (dataArrayData.length > 0) {

                        YIELD_CFM = (dataArray[i].QTY_CFM_IN / dataArrayData[0].QTY_CFM_OT_SUM) * 100;
                        dataArray[i].YIELD_CFM = YIELD_CFM;
                        ucProductRealgrid_gridView.setValue(i, "YIELD_CFM", dataArray[i].YIELD_CFM);
                    }
                }

                ucProductRealgrid_gridView.commit(true);

            } catch (e) {
                var test = e;
            }
        }

        //출고/입고 확정량, 출고/입고 차이량 계산
        function GridProdSetValue(Data, Value) {

            try {
                let DIGIT = 0.01;
                //목표수율
                let YIELD_GOAL = 0;
                //배부비율
                let RATIO_OT = 0;
                let QTY_OT_DIFF = 0;
                let QTY_CFM_OT = 0;

                ucProductRealgrid_gridView.commit(true);
                var dataArray = ucProductRealgrid_dataProvider.getJsonRows(0, -1);
                var Items_OT = [];

                for (var i = 0; i < dataArray.length; i++) {
                    if (Data.EQSGID == dataArray[i].EQSGID
                        && Data.PROCID == dataArray[i].PROCID
                        && Data.PRODID == dataArray[i].PRODID
                    ) {

                        if (dataArray[i].IN_EDIT == 'Y') { // 입고 보정 여부
                            dataArray[i].QTY_IN_DIFF = Value;
                            dataArray[i].QTY_CFM_IN = dataArray[i].QTY_IN_DIFF + dataArray[i].QTY_ERP_IN;

                            var dataArrayData = $.grep(dataArray, function (el, l) {
                                return el.EQSGID_OT == dataArray[i].EQSGID && el.PROCID_OT == dataArray[i].PROCID && el.MTRLID == dataArray[i].PRODID;
                            });

                            let QTY_IN_DIFF_SUM = 0;
                            if (dataArrayData.length > 0) {

                                for (var j = 0; j < dataArrayData.length; j++) {
                                    if (dataArrayData[j].QTY_OT_DIFF != undefined && !isNaN(dataArrayData[j].QTY_OT_DIFF)) {
                                        QTY_IN_DIFF_SUM += dataArrayData[j].QTY_OT_DIFF;
                                    }
                                }
                            } else {
                                QTY_IN_DIFF_SUM = dataArray[i].QTY_IN_DIFF;
                            }

                            if (dataArray[i].PROCID == 'HN6700' && dataArray[i].EQSGID == 'ES872') {

                                var test = QTY_IN_DIFF_SUM;

                            }

                            ucProductRealgrid_gridView.setValue(i, "QTY_IN_DIFF", QTY_IN_DIFF_SUM);

                            dataArray[i].QTY_CFM_IN = QTY_IN_DIFF_SUM + dataArray[i].QTY_ERP_IN;

                            if (dataArray[i].QTY_CFM_IN != undefined && !isNaN(dataArray[i].QTY_CFM_IN))
                                ucProductRealgrid_gridView.setValue(i, "QTY_CFM_IN", dataArray[i].QTY_CFM_IN);
                        }

                        if (dataArray[i].OT_EDIT == 'Y') {// 출고 보정 여부

                            //출고량/출고확정량 재계산
                            Items_OT.push(dataArray[i]);

                            RATIO_OT = dataArray[i].RATIO_OT;
                            YIELD_GOAL = dataArray[i].YIELD_GOAL;

                            if ((RATIO_OT != null && RATIO_OT != undefined)
                                && (YIELD_GOAL != null && YIELD_GOAL != undefined)) {

                                RATIO_OT *= DIGIT;
                                YIELD_GOAL *= DIGIT;

                                if (YIELD_GOAL <= 0) YIELD_GOAL = 1;

                                //let OT_QTY = 0;

                                //OT_QTY = dataArray[i].QTY_ERP_OT;

                                if (RATIO_OT == 1) {
                                    OT_QTY = dataArray[i].QTY_ERP_OT_SUM;
                                } else {
                                    OT_QTY = dataArray[i].QTY_ERP_OT;
                                }

                                //출고차이량 계산
                                if (dataArray[i].QTY_CFM_IN <= 0) {
                                    QTY_OT_DIFF = 0;
                                } else {
                                    QTY_OT_DIFF = (dataArray[i].QTY_CFM_IN / YIELD_GOAL.toFixed(10) * RATIO_OT) - OT_QTY;
                                }

                                /*QTY_OT_DIFF = (dataArray[i].QTY_CFM_IN / YIELD_GOAL.toFixed(10) * RATIO_OT) - OT_QTY;*/

            let retunValue = true;
            let checkTotalCount = 0;
            for (var i = 0; i < ucPackRealgrid_dataProvider.getRowCount(); i++) {
                var currentJsonRow = ucPackRealgrid_dataProvider.getJsonRow(i);
                if (SECTION == 'SEND') {
                    if (currentJsonRow.ERP_MTRL_OT_CHECK == true) {
                        checkTotalCount++;
                        if (currentJsonRow.ERP_MTRL_OT == 'S') {
                            xAlert('<%=lang.message["28214"]%>');  //ERP 전송 대상이 아닙니다.
                            return retunValue = false;
                            break;

                        if (currentJsonRow.QTY_OT_COR == 0) {

                            return retunValue = false;
                            break;
                        }
                    }
                } else {
                    if (currentJsonRow.ERP_MTRL_OT_CHECK == true) {
                        checkTotalCount++;
                        if (currentJsonRow.ERP_MTRL_OT != 'S') {
                            xAlert('<%=lang.message["28215"]%>');  //ERP 전송 취소 대상이 아닙니다.

                            break;

        //ERP 상태 체크 - 작업별 오더 배부
        function ErpSendProdValidation(SECTION) {

            let retunValue = true;
            let checkTotalCount = 0;
            for (var i = 0; i < ucWorkorderRealgrid_dataProvider.getRowCount(); i++) {
                var currentJsonRow = ucWorkorderRealgrid_dataProvider.getJsonRow(i);

                if (SECTION == 'SEND') {
                    //ERP 전송
                    //ERP 전송 성공 상태는 제외

                    if (currentJsonRow.ERP_MTRL_STATUS_CHECK == true) { // 출고 체크
                        checkTotalCount++;

                        if (currentJsonRow.QTY_OT_COR == undefined || currentJsonRow.QTY_OT_COR == 0) {//전송량 체크

                            xAlert('<%=lang.message["25169"]%>');  //ERP 전송하는 수량이 없습니다.
                            return retunValue = false;
                            break;
                        }
                    }

                    if (currentJsonRow.ERP_PROD_STATUS_CHECK == true) {// 입고 체크
                        checkTotalCount++;

                        if (currentJsonRow.ERP_PROD_STATUS == 'N') {
                            xAlert('<%=lang.message["28214"]%>');  //ERP 전송 대상이 아닙니다.
                            return retunValue = false;
                            break;
                        }

                        if (currentJsonRow.QTY_IN_COR == undefined || currentJsonRow.QTY_IN_COR == 0) {//전송량 체크
                            xAlert('<%=lang.message["25169"]%>');  //ERP 전송하는 수량이 없습니다.
                            return retunValue = false;
                            break;
                        }
                    }
                } else {
                    //ERP 전송취소
                    //ERP 전송 성공 상태만 허용

                    if (currentJsonRow.ERP_MTRL_STATUS_CHECK == true) {
                        checkTotalCount++;
                        if (currentJsonRow.ERP_MTRL_STATUS != 'S') {
                            xAlert('<%=lang.message["28215"]%>');  //ERP 전송 취소 대상이 아닙니다.
                            return retunValue = false;
                            break;
                        }
                    }

                    if (currentJsonRow.ERP_PROD_STATUS_CHECK == true) {
                        checkTotalCount++;
                        if (currentJsonRow.ERP_PROD_STATUS != 'S') {
                            xAlert('<%=lang.message["28215"]%>');  //ERP 전송 취소 대상이 아닙니다.
                            return retunValue = false;
                            break;
                        }
                    }
                }
            }

            if (retunValue == true) {
                if (checkTotalCount <= 0) {
                    xAlert('<%=lang.message["28216"]%>');  //ERP 전송 대상이 존재하지 않습니다.
                    return retunValue = false;
                }
            }

            return retunValue;
        }

        function GetConvertMulti(arrLotItems) {

            var TextList = '';

            if (arrLotItems.length > 0) {

                var tempItems = arrLotItems.split(',');

                for (var i = 0; i < tempItems.length; i++) {
                    TextList = TextList.concat("'", tempItems[i].trim(), "',");
                };
                TextList = TextList.concat("''");
            };
            return TextList;
        };

        function CallAjaxPOST(_url) {
            var deferred = $.Deferred();
            $.ajaxSetup({ contentType: "application/json; charset=utf-8", data: JSON.stringify(null), dataType: "json" });
            $.ajax({
                method: "POST",
                url: _url,
                dataType: "json",
                cache: false,
                success: function (result) {
                    deferred.resolve(result);
                },
                error: function (result) {
                    alert(result);
                    deferred.resolve(null);
                }
            });

            return deferred.promise();
        };

            ucWipRealgrid_gridView.onColumnCheckedChanged = function (grid, col, chk) {
                //hearder check
                let columnName = col.name;
                let bChecked = chk;

                ucWipRealgrid_dataProvider.getJsonRows(0, -1).forEach(function (value, index, array) {

                    if (columnName == "ERP_MTRL_IN_CHECK") {

                        if (bChecked == false) {
                            grid.setValue(index, columnName, bChecked);
                        }

                        if (value.LOTID_PRE != "") {
                            if (value.INOUT.indexOf("IN") > -1
                                && value.PROCWORKRESULT_REAL == ''
                                && value.QTY_WIP_PRE > 0) {
                                grid.setValue(index, columnName, !bChecked);
                                grid.setValue(index, columnName, bChecked);
                            }
                        }

                    } else if (columnName == "ERP_MTRL_OT_CHECK" || columnName == "ERP_MOVE_OT_CHECK") {

                        if (bChecked == false) {
                            grid.setValue(index, "ERP_MTRL_OT_CHECK", bChecked);
                            grid.setValue(index, "ERP_MOVE_OT_CHECK", bChecked);
                        }

                        if (value.INOUT.indexOf("OUT") > -1
                            && value.PROCWORKRESULT_REAL == ''
                            && value.QTY_WIP_REAL > 0) {

                            grid.setValue(index, "ERP_MTRL_OT_CHECK", !bChecked);
                            grid.setValue(index, "ERP_MOVE_OT_CHECK", !bChecked);

                            grid.setValue(index, "ERP_MTRL_OT_CHECK", bChecked);
                            grid.setValue(index, "ERP_MOVE_OT_CHECK", bChecked);
                        } else {
                            if (value.PROCWORKRESULT_REAL == '') {
                                if (value.ERP_MTRL_OT.indexOf('S') > -1
                                    || value.ERP_MTRL_OT.indexOf('E') > -1
                                ) {
                                    grid.setValue(index, "ERP_MTRL_OT_CHECK", !bChecked);
                                    grid.setValue(index, "ERP_MTRL_OT_CHECK", bChecked);
                                }

                                if (value.ERP_MOVE_OT.indexOf('S') > -1
                                    || value.ERP_MOVE_OT.indexOf('E') > -1
                                ) {
                                    grid.setValue(index, "ERP_MOVE_OT_CHECK", !bChecked);
                                    grid.setValue(index, "ERP_MOVE_OT_CHECK", bChecked);
                                }
                            }
                        }
                    }

                    grid.commit(true);
                });

                ERP_WIP_SendText();
            };

        //포장 실적 계산
        function SetPackBagCalc(itemIndex) {

            ucPackRealgrid_gridView.commit(true);

            var current = ucPackRealgrid_gridView.getValues(itemIndex);

            if (current != undefined) {

                var dataArray = ucPackRealgrid_dataProvider.getJsonRows(0, -1);
                let QTY_OT_CFM_SUM = current.QTY_OT_CFM_SUM;
                let QTY_OT_CFM_WO_SUM = 0;
                let QTY_OT_CFM_LIST = [];

                dataArray.forEach(function (value, index, array) {

                    let AREAID = ucPackRealgrid_gridView.getValue(index, 'AREAID');
                    let MTRLID = ucPackRealgrid_gridView.getValue(index, 'MTRLID');
                    let ERP_MTRL_OT = ucPackRealgrid_gridView.getValue(index, 'ERP_MTRL_OT');

                    if (current.MTRLID == MTRLID) {
                        //WOID
                        let WOID = ucPackRealgrid_gridView.getValue(index, 'WOID');
                        //확정량
                        let QTY_OT_CFM = ucPackRealgrid_gridView.getValue(index, 'QTY_OT_CFM');
                        //상태
                        let ERP_MTRL_OT = ucPackRealgrid_gridView.getValue(index, 'ERP_MTRL_OT');

                        //생산비율
                        let RATIO_BAG = ucPackRealgrid_gridView.getValue(index, 'RATIO_BAG');
                        //확정량(오더별)
                        let QTY_OT_CFM_WO = (QTY_OT_CFM * RATIO_BAG).toFixed(0);

                        QTY_OT_CFM_WO_SUM += Number(QTY_OT_CFM_WO);

                        ucPackRealgrid_gridView.setValue(index, "QTY_OT_CFM_WO", QTY_OT_CFM_WO);
                        //기전송 전송량
                        let QTY_OT = ucPackRealgrid_gridView.getValue(index, 'QTY_OT');
                        //보정량(261,262)
                        let QTY_OT_COR = QTY_OT_CFM_WO - QTY_OT;
                        ucPackRealgrid_gridView.setValue(index, "QTY_OT_COR", QTY_OT_COR);

                        QTY_OT_CFM_LIST.push({
                            WOID: WOID
                            , AREAID: AREAID
                            , MTRLID: MTRLID
                            , QTY_OT_CFM_WO: QTY_OT_CFM_WO
                            , RATIO_BAG: RATIO_BAG
                        });
                    }
                });



                //기전송 ERP 전송량,확정량(오더별) 차이량 재계산
                //생산 비율이 제일 높은 오더에 차이량을 더해 준다.
                if (current.QTY_OT_CFM != QTY_OT_CFM_WO_SUM) {

                    let QTY_OT_DIFF = current.QTY_OT_CFM - QTY_OT_CFM_WO_SUM;

                    const maxTaget = QTY_OT_CFM_LIST.reduce(function (prev, current) {
                        return (prev && prev.RATIO_BAG > current.RATIO_BAG) ? prev : current
                    });

                    let MDATA = maxTaget;

                    dataArray.forEach(function (value, index, array) {

                        let AREAID = ucPackRealgrid_gridView.getValue(index, 'AREAID');
                        let PROCID = ucPackRealgrid_gridView.getValue(index, 'PROCID');
                        let EQSGID = ucPackRealgrid_gridView.getValue(index, 'EQSGID');
                        let MTRLID = ucPackRealgrid_gridView.getValue(index, 'MTRLID');
                        let WOID = ucPackRealgrid_gridView.getValue(index, 'WOID');


                        if (MDATA.MTRLID == MTRLID && MDATA.WOID == WOID) {

                            let QTY_OT_CFM_WO = ucPackRealgrid_gridView.getValue(index, 'QTY_OT_CFM_WO');

                            QTY_OT_CFM_WO += Number(QTY_OT_DIFF);

                            ucPackRealgrid_gridView.setValue(index, "QTY_OT_CFM_WO", QTY_OT_CFM_WO);
                            //기전송 전송량
                            let QTY_OT = ucPackRealgrid_gridView.getValue(index, 'QTY_OT');
                            //보정량(261,262)
                            let QTY_OT_COR = QTY_OT_CFM_WO - QTY_OT;
                            ucPackRealgrid_gridView.setValue(index, "QTY_OT_COR", QTY_OT_COR);
                        }
                    });
                }

                // ERP 전송 체크박스
                // 보정량에 따라 체크 박스 체크
                dataArray.forEach(function (value, index, array) {

                    let QTY_OT_COR = ucPackRealgrid_gridView.getValue(index, 'QTY_OT_COR');

                    let ORG_ERP_MTRL_OT = ucPackRealgrid_gridView.getValue(index, 'ORG_ERP_MTRL_OT');

                    let ERP_MTRL_OT = ucPackRealgrid_gridView.getValue(index, 'ERP_MTRL_OT');

                    ucPackRealgrid_gridView.setValue(index, "ERP_MTRL_OT", ORG_ERP_MTRL_OT);

                    ucPackRealgrid_gridView.setCellStyle(index, "ERP_MTRL_OT_CHECK", "readonlyCellCheckBoxStyle", true);
                    ucPackRealgrid_gridView.setValue(index, "ERP_MTRL_OT_CHECK", true);

                    if (QTY_OT_COR != undefined && QTY_OT_COR != 0) {
                        ucPackRealgrid_gridView.setCellStyle(index, "ERP_MTRL_OT_CHECK", "editCellCheckBoxStyle", true);
                    } else {

                        ucPackRealgrid_gridView.setValue(index, "ERP_MTRL_OT_CHECK", false);

                        if (ERP_MTRL_OT == "S" || ERP_MTRL_OT == "E") {
                            ucPackRealgrid_gridView.setCellStyle(index, "ERP_MTRL_OT_CHECK", "editCellCheckBoxStyle", true);
                            ucPackRealgrid_gridView.setValue(index, "ERP_MTRL_OT_CHECK", true);
                        } else {
                            if (QTY_OT_COR != undefined && QTY_OT_COR == 0) {
                                ucPackRealgrid_gridView.setCellStyle(index, "ERP_MTRL_OT_CHECK", "readonlyCellCheckBoxStyle", true);
                                ucPackRealgrid_gridView.setValue(index, "ERP_MTRL_OT", 'M');
                            }
                        }
                    }
                });
            }

            ucPackRealgrid_gridView.commit(true);
            ErpPackBagSendText();
        }

        //작업오더 배부 ERP 전송 대상 정리
        function ERP_PROD_SendText() {
            ucWorkorderRealgrid_gridView.commit(true);

            let cnt_261 = 0;
            let cnt_262 = 0;
            let cnt_101 = 0;
            let cnt_102 = 0;


            let PROD_LOTID = '';
            let PROD_WOID = '';
            for (var i = 0; i < ucWorkorderRealgrid_dataProvider.getRowCount(); i++) {
                var currentJsonRow = ucWorkorderRealgrid_dataProvider.getJsonRow(i);

                if (currentJsonRow.ERP_MTRL_STATUS_CHECK == true) { // 출고 체크
                    if (currentJsonRow.QTY_OT_COR != undefined && currentJsonRow.QTY_OT_COR != 0) {//전송량 체크

                        if (currentJsonRow.QTY_OT_COR > 0)
                            cnt_261++;
                        else
                            cnt_262++;
                    }
                }

                if (currentJsonRow.ERP_PROD_STATUS_CHECK == true) {// 입고 체크
                    if (currentJsonRow.QTY_IN_COR != undefined && currentJsonRow.QTY_IN_COR != 0) {//전송량 체크


                        if (PROD_LOTID == '' || PROD_WOID == '') {
                            PROD_LOTID = currentJsonRow.LOTID;
                            PROD_WOID = currentJsonRow.WOID;
                        } else {

                            if (PROD_LOTID == currentJsonRow.LOTID && PROD_WOID == currentJsonRow.WOID) {
                                continue;
                            } else {
                                PROD_LOTID = currentJsonRow.LOTID;
                                PROD_WOID = currentJsonRow.WOID;
                            }
                        }

                        if (currentJsonRow.QTY_IN_COR > 0)
                            cnt_101++;
                        else
                            cnt_102++;
                    }
                }
            }

            var innerHtml = '';
            innerHtml += "261(" + "<span class='red01'>" + cnt_261 + "</span>" + "), 262(" + "<span class='red01'>" + cnt_262 + "</span>" + "), 101(" + "<span class='red01'>" + cnt_101 + "</span>" + "),102(" + "<span class='red01'>" + cnt_102 + "</span>" + ")";

            $("#divErpProdSndText").html(innerHtml);

            ucWorkorderRealgrid_LoadDataCompleted();
        }

                        //전송 취소후 변경 가능
                        let MTRLLIST = $.grep(ERP_MTRL_OT_LIST, function (el, l) { return el.MTRLID == MTRLID; });

                            ucPackRealgrid_gridView.setCellStyle(index, "QTY_OT_CFM", "readonlyCellStyle", true);
                        } else {
                            ucPackRealgrid_gridView.setCellStyle(index, "QTY_OT_CFM", "editCellStyle", true);

        //ERP 상태 아이콘 클릭시 체크(포장 실적) 
        function SetERP_PACK_BAG_WIP(img, STATUS) {
            ucPackRealgrid_gridView.commit(true);

            var celleditStyle = 'editCellCheckBoxStyle';

            for (var i = 0; i < ucPackRealgrid_dataProvider.getRowCount(); i++) {
                var currentJsonRow = ucPackRealgrid_dataProvider.getJsonRow(i);

                ucPackRealgrid_gridView.setValue(i, "ERP_MTRL_OT_CHECK", false);

                if (STATUS == "S" || STATUS == "E") {
                    if (currentJsonRow.ERP_MTRL_OT == STATUS)
                        ucPackRealgrid_gridView.setValue(i, "ERP_MTRL_OT_CHECK", true);
                } else {
                    if (currentJsonRow.ERP_MTRL_OT == STATUS
                        && currentJsonRow.QTY_OT_COR != undefined && currentJsonRow.QTY_OT_COR != 0
                    ) {
                        ucPackRealgrid_gridView.setValue(i, "ERP_MTRL_OT_CHECK", true);
                    }
                }
            }

            ucPackRealgrid_gridView.commit(true);
            ErpPackBagSendText();
        }


        //포장 실적 ERP 전송대상 정리
        function ErpPackBagSendText() {
            ucPackRealgrid_gridView.commit(true);

            let cnt_261 = 0;
            let cnt_262 = 0;

            for (var i = 0; i < ucPackRealgrid_dataProvider.getRowCount(); i++) {
                var currentJsonRow = ucPackRealgrid_dataProvider.getJsonRow(i);

                if (currentJsonRow.ERP_MTRL_OT_CHECK == true) { // 출고 체크
                    if (currentJsonRow.QTY_OT_COR != undefined && currentJsonRow.QTY_OT_COR != 0) {//전송량 체크

                        if (currentJsonRow.QTY_OT_COR > 0)
                            cnt_261++;
                        else
                            cnt_262++;
                    }
                }
            }

            var innerHtml = '';
            innerHtml += "261(" + "<span class='red01'>" + cnt_261 + "</span>" + "), 262(" + "<span class='red01'>" + cnt_262 + "</span>" + ")";

            $("#divErpPackBagSndText").html(innerHtml);

            ucPackRealgrid_LoadDataCompleted();
        }
