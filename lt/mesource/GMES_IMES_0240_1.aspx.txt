        function formatter2(date) {
            if (!date) { return ''; }
            var y = date.getFullYear();
            var m = date.getMonth() + 1;
            return y + '-' + (m < 10 ? ('0' + m) : m);
        }
        function parser2(s) {
            if (!s) { return null; }
            var ss = s.split('-');
            var y = parseInt(ss[0], 10);
            var m = parseInt(ss[1], 10);
            if (!isNaN(y) && !isNaN(m)) {
                return new Date(y, m - 1, 1);
            } else {
                return new Date();
            }
        }

        function SetMeasureItem() {
            var param1 = 'MeasureItem';
            var param2 = "'CA', 'EE', 'MS', 'OA', 'PC', 'PQ', 'SD'";
            $('#cboMeasureItem').combobox({
                //url: '../common/xml/CallBizJson.aspx?sp_name=BR_IM_COM_GET_COMMONCODE_CBO&LANGID=' + XSSReplace( $("[id$=hidLangID]").val()  , 1) + '&CMCDTYPE=MeasureItem&CBOOPT=OPT|CMCODE|CMCDNAME',
                url: '../common/xml/CallBizJson.aspx?sp_name=BR_CUS_SEL_COMMONCODE_CBO_CMCODE&LANGID=' + XSSReplace( $("[id$=hidLangID]").val()  , 1) + '&CMCDTYPE=' + param1 + '&CMCODE=' + param2 + '&CBOOPT=OPT|CMCODE|CMCDNAME',
                valueField: 'CMCODE',
                textField: 'CMCDNAME',
                onLoadSuccess: function () {
                    var items = $(this).combobox("getData");
                    if (items.length === 2) {
                        var opts = $(this).combobox("options");
                        $(this).combobox("select", items[1][opts.valueField]);
                    }
                },
                onLoadSuccess: function () {
                    var autoSelect = false;
                    var MeasureItem = $("[id$=hidMeasureID]").val();
                    if ($(this).combobox("getData").length > 0) {
                        var items = $(this).combobox("getData");
                        var opts = $(this).combobox("options");
                        var strIn = false;
                        for (var i = 0; i < items.length; i++) {
                            if (items[i][opts.valueField] == MeasureItem) { // 콤보박스 데이터가 있다면 있음 표시 하고 SELECT될 수 있도록 한다.
                                strIn = true;
                                break;
                            }
                        }
                        if (strIn) {
                            $(this).combobox("select", MeasureItem);
                        } else {
                            autoSelect = true;
                        }
                    } else {
                        autoSelect = true;
                    }

                    if (autoSelect) {

                        var items = $(this).combobox("getData");
                        if (items.length === 2) {
                            var opts = $(this).combobox("options");
                            $(this).combobox("select", items[1][opts.valueField]);
                        }
                    }
                }
            });

        function getFormatDate(date) {
            var dateArray = date.split('/');
            var year = '20' + dateArray[2];              //yyyy
            var month = dateArray[0];
            month = month >= 10 ? month : '0' + month;  //month 두자리로 저장
            var day = dateArray[1];
            day = day >= 10 ? day : '0' + day;          //day 두자리로 저장
            return year + '-' + month + '-' + day;
        }

        function getFormatInt(num) {
            var returnNum;
            if (num == ' - ') { // 복붙해서 Excel 입력하는 중에, 0인 값 -> ' - '로 저장 되는 경우
                returnNum = 0;
            } else if (num == '' || num == null) { // 입력 안한 경우
                returnNum = 0;
            } else { // 숫자 양식 ',' 포함 경우 
                returnNum = num.replace(/\,/g, '');
            }
            return returnNum;
        }

function ButtonClick_Delete(obj) {

    var chkRows = UCRealgrid_gridView.getCheckedRows();

    for (var i = chkRows.length - 1; i >= 0; i--) {

        if (UCRealgrid_dataProvider.getRowState(chkRows[i]) === "created") {
            UCRealgrid_dataProvider.setOptions({ softDeleting: false });
        }
        else {
            UCRealgrid_dataProvider.setOptions({ softDeleting: true });
            UCRealgrid_gridView.setOptions({ hideDeletedRows: true });
        }
        UCRealgrid_dataProvider.removeRow(chkRows[i]);
    };
    UCRealgrid_gridView.commit(true);
    var rowNo = UCRealgrid.GetRowCount();
};
