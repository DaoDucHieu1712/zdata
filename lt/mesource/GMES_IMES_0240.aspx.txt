        function formatter2(date) {
            if (!date) { return ''; }
            var y = date.getFullYear();
            var m = date.getMonth() + 1;
            return y + '-' + (m < 10 ? ('0' + m) : m);
        }
        function parser2(s) {
            if (!s) { return null; }
            var ss = s.split('-');
            var y = parseInt(ss[0], 10);
            var m = parseInt(ss[1], 10);
            if (!isNaN(y) && !isNaN(m)) {
                return new Date(y, m - 1, 1);
            } else {
                return new Date();
            }
        }

        function setCloseMonth() {
            var toDay = $.fn.datebox.defaults.formatter(new Date());

            var WipCloseDate;
            var WipClose;
            // <summary>실사일자에 해당되는 마감월을 설정한다.</summary> 
            var items = {};
            var dtDate = $('#dtDate').datetimespinner('getValue');
            items.WIPCLOSEMONTH = dtDate.replace('-', '');
            items.MENUID = 'IUSM411001';
            items.SHOPID = $("[id$=hidShopID]").val();

            var param = {};
            param.bizID = "DA_IM_PRD_SEL_WIPCLOSE";
            param.items = items;

            var url = "/GMES_COM/Service/CallBizJson.aspx/GetBizJsonByDictionary";

            sendRequestMethod(function (id, datas) {
                if (datas != null && datas.length > 0) {
                    WipCloseDate = datas[0].WIPCLOSEDATE;
                    if (new Date(toDay) <= new Date(WipCloseDate)) {//WipClose 이전
                        SetButtonEnable("#btnSave", false);
                        SetButtonEnable("#btnRowAdd", true);
                        SetButtonEnable("#btnRowDelete", true);
                        SetButtonEnable("#btnDel", true);
                    } else {
                        SetButtonEnable("#btnSave", false);
                        SetButtonEnable("#btnRowAdd", false);
                        SetButtonEnable("#btnRowDelete", false);
                        SetButtonEnable("#btnDel", false);
                    }

                    //session에 wipclose값 저장
                    WipClose = datas[0].WIPCLOSE;
                    //sessionStorage
                    window.sessionStorage.clear();
                    window.sessionStorage.setItem("WipClose", WipClose);
                    //조회를 누른 시점의 마감월을 session에 저장
                    window.sessionStorage.setItem("dtDate", $('#dtDate').datetimespinner('getValue'));

                }
                vFirst = false;
            }, param, "POST", url);


        }

        function UCRealgrid_KeyDown(grid, key, ctrl, shift, alt) {

                        gridSeq = UCRealgrid_gridView.getCurrent().dataRow;
                        UCRealgrid_gridView.commit(true);

                        if (gPRODID != undefined) {
                            if (gPRODID.length > 0) {

                                ShowProductPopup(gridSeq, gridSeq, gPRODID);

                                //GetProdData(gPRODID);
                                //ShowProductPopup(gPRODID);
                            }
                        }
                    }
                    UCRealgrid_gridView.commit(true);
                }
            }
        }

        function SetProductName(data) {
            if (data == undefined) return;
            //if (data.length != 3) return;
            if (data.length < 3) return; //2022.05.26 황유라 txtProductGroup 객체 없음, 주석처리

            //data[0];//Part No
            //data[1];//제품명
            //data[2];//제품코드

            var current = UCRealgrid_gridView.getCurrent();
            var row = current.dataRow;
            var field = current.fieldIndex;


            //제품 입력 도중 입력 커서가 바뀌어도 정확한 위치에 기입될 수 있도록 위치 강제
            if ($('#cboMeasureItem').combobox('getValue') == "MS") {
                //var productField = 8;
                var productField = 9;
                field = productField;
            }
            else if ($('#cboMeasureItem').combobox('getValue') == "CA") {
                var productField = 9;
                field = productField;
            }
            if ($('#cboMeasureItem').combobox('getValue') == "MS") {
                UCRealgrid_dataProvider.setValue(row, field, data[2]);
                UCRealgrid_dataProvider.setValue(row, field + 1, data[1]);
            }
            else if ($('#cboMeasureItem').combobox('getValue') == "CA") {
                UCRealgrid_dataProvider.setValue(row, field, data[2]);
                UCRealgrid_dataProvider.setValue(row, field + 1, data[1]);

                UCRealgrid_dataProvider.setValue(row, field - 2, data[2]);
                UCRealgrid_dataProvider.setValue(row, field - 1, data[1]);
            }
            else {
                UCRealgrid_dataProvider.setValue(row, field - 1, data[2]);
                UCRealgrid_dataProvider.setValue(row, field, data[1]);
            }
        }

        function ButtonClick_Delete(obj) {

            var chkRows = UCRealgrid_gridView.getCheckedRows();

            for (var i = chkRows.length - 1; i >= 0; i--) {

                if (UCRealgrid_dataProvider.getRowState(chkRows[i]) === "created") {
                    UCRealgrid_dataProvider.setOptions({ softDeleting: false });
                }
                else {
                    UCRealgrid_dataProvider.setOptions({ softDeleting: true });

                }
                UCRealgrid_dataProvider.removeRow(chkRows[i]);
            };
            UCRealgrid_gridView.commit(true);
            searchData();
        }
