            var gridMaster = document.getElementById("ucMasterRealgrid");

                };

            i = pageHeight - masterHeight - dockh - 40;

        function SetDateTime() {
            var toDate = new Date();
            var fromDate = new Date();
            var dayOfMonth = fromDate.getDate();
            fromDate.setDate(dayOfMonth - 0);
            toDate.setDate(dayOfMonth);

            toDate = new Date(toDate.getFullYear(), toDate.getMonth(), toDate.getDate());

            $('#dtDateRange').daterangebox('SetFromDate', fromDate);
            $('#dtDateRange').daterangebox('SetToDate', toDate);

        function CallAjaxPOST(_url) {
            var deferred = $.Deferred();
            $.ajaxSetup({ contentType: "application/json; charset=utf-8", data: JSON.stringify(null), dataType: "json" });
            $.ajax({
                method: "POST",
                url: _url,
                dataType: "json",
                cache: false,
                success: function (result) {
                    deferred.resolve(result);
                },
                error: function (result) {
                    alert(result);
                    deferred.resolve(null);
                }
            });

            return deferred.promise();
        }

                    };
